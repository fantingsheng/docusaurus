import Link from '@docusaurus/Link';

const TerminalLine = ({ children, blinking = false }) => (
 <div>

  <span style={{ fontStyle: 'italic' }}>{children}</span>
 </div>
);

const TerminalResponse = ({ children }) => <div style={{ lineHeight: '1.5rem' }}>{children}</div>;

 return (

    </TerminalLine>

    <TerminalResponse>
