## 用户信息

Github 提供了 API 可查看用户的大部分可公开的信息

在下面输入框中输入用户名，点击查看

export function GithubUserInfo() {
  const [userName, setUserName] = useState('fantingsheng');
  return (
    <Box sx={{ margin: '40px 0' }}>
      <Box sx={{ width: '50%' }}>
        <TextField
          label='Github 用户名'
          value={userName}
          onChange={e => setUserName(e.target.value)}
          placeholder='Github User Name'
          size='small'
          fullWidth
        />
      </Box>
      <Box sx={{ marginTop: '15px' }}>
        <Typography variant="caption" display="block" gutterBottom>
          {`https://api.github.com/users/${userName}`}
        </Typography>
        <a href={`https://api.github.com/users/${userName}`} target='_blank'>
          点击查看
        </a>
      </Box>
    </Box>
  )
}

## 用户头像

Github 账号的用户 ID 可通过上面的链接在用户信息里拿到。
或者通过网站 [https://caius.github.io/github_id](https://caius.github.io/github_id) 直接查询

export function GithubAvatar () {
  const [copied, setCopied] = useState(false);
  const [config, setConfig] = useState({
    userId: '6924543',
    size: 128
  });
  const handleFormChange = (val, key) => {
    setConfig((prev) => ({ ...prev, [key]: val }));
  };
  const onCopyHandle = (iconText) => {
    const link = `https://avatars.githubusercontent.com/u/${config.userId}?v=4&size=${config.size}`
    navigator.clipboard.writeText(link)
    .then(() => {
      setCopied(true);
      console.log("已成功复制到剪贴板");
    })
    .catch((error) => {
      console.error("复制时出错: ", error);
    });
  }
  return (
    <Box sx={{ margin: '40px 0' }}>
      <Grid container spacing={4}>
        <Grid item xs={4}>
          <Box>
            <Typography sx={{ }} gutterBottom>
              用户 ID
            </Typography>
            <TextField
              value={config.userId}
              onChange={e => handleFormChange(e.target.value, 'userId')}
              placeholder='Github ID'
              size='small'
              fullWidth
            />
          </Box>
        </Grid>
        <Grid item xs={4}>
          <Box>
            <Typography sx={{ }} gutterBottom>
              尺寸
            </Typography>
            <ToggleButtonGroup
              value={config.size}
              exclusive
              onChange={(e, val) => handleFormChange(val, 'size')}
              size='small'
            >
              <ToggleButton value={64}>64px</ToggleButton>
              <ToggleButton value={128}>128px</ToggleButton>
              <ToggleButton value={192}>192px</ToggleButton>
              <ToggleButton value={256}>256px</ToggleButton>
              <ToggleButton value={300}>300px</ToggleButton>
            </ToggleButtonGroup>
          </Box>
        </Grid>
      </Grid>
      <Avatar
        alt="Github Avatar"
        sx={{ width: config.size, height: config.size, margin: '20px 0' }}
        src={`https://avatars.githubusercontent.com/u/${config.userId}?v=4&size=${config.size}`}
      />
      <Box sx={{ marginTop: '20px' }}>
        <Typography variant="caption" display="block" gutterBottom>
          {`https://avatars.githubusercontent.com/u/${config.userId}?v=4&size=${config.size}`}
        </Typography>
        <Button variant="text" onClick={onCopyHandle} disableElevation>复制链接</Button>
      </Box>
      <Snackbar open={copied} autoHideDuration={1500} anchorOrigin={{ horizontal: 'center', vertical: 'bottom' }} onClose={() => setCopied(false)}>
        <Alert severity="success"><AlertTitle>复制成功</AlertTitle></Alert>
      </Snackbar>
    </Box>
  )
}
