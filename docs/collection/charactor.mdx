# 网页特殊表情和字符

文档中搭配一些表情字符会使得文字**生动可爱**，收集了一些经常使用的字符图标。

import { Box, Grid }  from '@mui/material';
export const EmojiGallery = ({ emojiList }) => {
  return (
    <Box sx={{ flexGrow: 1 }}>
      <Grid container spacing={2}>
        {
          emojiList.map((el, index) => (
            <Grid item xs={2} md={1} key={index}>
              <div style={{ textAlign: 'center', fontSize: '3rem', width: '100%', borderRadius: '10px' }}>{el}</div>
            </Grid>
          ))
        }
      </Grid>
    </Box>
  )
}
