## 树形结构增加属性

如果我们要给菜单的每一项增加一个层级的属性，则需要通过递归的方式，实现下面这样一个构造的方法：

```js showLineNumbers
const arrayTreeAddLevel = (array, levelName = 'level', childrenName = 'children') => {
  if (!Array.isArray(array)) return [];
  const recursive = (array, level = 0) => {
    level++;
    return array.map(v => {
      v[levelName] = level;
      const child = v[childrenName];
      if (child && child.length) recursive(child, level);
      return v;
    })
  }
  return recursive(array)
}
```

然后，我们将树形结构的数据通过参数传入调用后：

```js showLineNumbers
arrayTreeAddLevel(treeData);
```

数据经过处理后，输出如下：

## 树形结构拍平

如果是将树形结构的数据拍平，返回一个一维数组呢？

```js showLineNumbers
export function flatTree(data, children?) {
  const arrList = [];
  const config = {
    childrenList: children || "children"
  };

  for (const d of data) {
    adaptToChildrenList(d);
  }

  function adaptToChildrenList(o) {
    const obj = Object.entries(o).reduce(
      (a, [k, v]) => (k === config.childrenList ? a : ((a[k] = v), a)),
      {}
    );
    arrList.push(obj);
    if (o[config.childrenList] && o[config.childrenList].length > 0) {
      for (const l of o[config.childrenList]) {
        adaptToChildrenList(l);
      }
    }
  }
  return arrList;
}
```

## 按照某个字段排序

在菜单的使用场景中，我们需要对同级的数组按照某个字段进行排序：

```js showLineNumbers
function sortTree(data, attr = 'rank') {
  const menus = data.sort(
    (a, b) => {
      return a[attr] - b[attr];
    }
  );
  menus.forEach(
    v => v.children && (v.children = sortTree(v.children))
  );
  return menus;
}
```

通过切换按钮，可直观观察排序前后的数据变化。

export const sortTree = (data, attr = 'rank') => {
  const menus = data.sort(
    (a, b) => {
      return a[attr] - b[attr];
    }
  );
  menus.forEach(
    v => v.children && (v.children = sortTree(v.children))
  );
  return menus;
}

export const ShowTreeSort = () => {
  const [sort, setSort] = useState(false);
  const originTree = [
    {
      id: 1, name: 'a', rank: 3, children: [
        { id: 101, name: 'a1', rank: 4, children: null },
        { id: 102, name: 'a2', rank: 2, children: null }
      ]
    },
    {
      id: 2, name: 'b', rank: 2, children: [
        {
          id: 201, name: 'b1', rank: 2, children: [
            { id: 211, name: 'b2', rank: 1, children: null }
          ]
        }
      ]
    }
  ];

  const data = sort ? sortTree(originTree) : originTree;

  return (
    <div>
      <Box sx={{ display: 'flex' }}>
          <FormControlLabel
            control={
              <Switch
                value={sort}
                onChange={e => setSort(e.target.checked)}
                inputProps={{ 'aria-label': 'controlled' }}
              />
            }
            label="开启排序"
          />
        </Box>
      <JsonView enableClipboard={false} src={data} />
    </div>
  )
}
